cmake_minimum_required(VERSION 3.16)
project(rampd VERSION 0.6.0 LANGUAGES CXX)
option(RAMPD_BUILD_TESTS "Build mandatory native regression tests" ON)
option(RAMPD_SANITIZE "Enable AddressSanitizer and UBSan (native tests)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(MPFR_INCLUDE_DIR mpfr.h HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include)
find_library(MPFR_LIBRARY NAMES mpfr HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES lib lib64)
find_library(GMP_LIBRARY NAMES gmp HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES lib lib64)
if(NOT MPFR_INCLUDE_DIR OR NOT MPFR_LIBRARY OR NOT GMP_LIBRARY)
  message(FATAL_ERROR "MPFR/GMP development files missing. Ubuntu/Debian: sudo apt-get install libmpfr-dev libgmp-dev. Or set CMAKE_PREFIX_PATH to a user-installed prefix. No Boost fallback is permitted.")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "This release supports GCC/Clang builds only.")
endif()
add_library(rampd_interval SHARED cpp/src/interval.cpp cpp/src/interval_bicycle.cpp cpp/src/picard.cpp)
target_include_directories(rampd_interval PUBLIC cpp/include PRIVATE ${MPFR_INCLUDE_DIR})
target_link_libraries(rampd_interval PRIVATE ${MPFR_LIBRARY} ${GMP_LIBRARY})
target_compile_options(rampd_interval PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
set_target_properties(rampd_interval PROPERTIES OUTPUT_NAME rampd_interval)
if(RAMPD_SANITIZE)
  target_compile_options(rampd_interval PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rampd_interval PUBLIC -fsanitize=address,undefined)
endif()
if(RAMPD_BUILD_TESTS)
  enable_testing()
  foreach(test interval picard abi)
    add_executable(test_${test} cpp/src/test_${test}.cpp)
    target_link_libraries(test_${test} PRIVATE rampd_interval)
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    add_test(NAME ${test}_cpp COMMAND test_${test})
  endforeach()
endif()
